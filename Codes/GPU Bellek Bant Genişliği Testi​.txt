#include <stdio.h>​

#include <cuda_runtime.h>​

#include <chrono>​

​

// Veri transfer süresini ölçme fonksiyonu​

void measure_transfer_time(size_t size, bool usePinnedMemory) {​

    int *h_data, *d_data;​

    cudaMalloc((void **)&d_data, size);​

​

    if (usePinnedMemory) {​

        cudaHostAlloc((void **)&h_data, size, cudaHostAllocDefault);  // Pinned memory​

    } else {​

        h_data = (int *)malloc(size);  // Normal malloc (Paged memory)​

    }​

​

    // CPU → GPU (H2D) transfer süresi​

    auto start = std::chrono::high_resolution_clock::now();​

    cudaMemcpy(d_data, h_data, size, cudaMemcpyHostToDevice);​

    cudaDeviceSynchronize();​

    auto end = std::chrono::high_resolution_clock::now();​

    double h2d_time = std::chrono::duration<double, std::milli>(end - start).count();​

    ​

    // GPU → CPU (D2H) transfer süresi​

    start = std::chrono::high_resolution_clock::now();​

    cudaMemcpy(h_data, d_data, size, cudaMemcpyDeviceToHost);​

    cudaDeviceSynchronize();​

    end = std::chrono::high_resolution_clock::now();​

    double d2h_time = std::chrono::duration<double, std::milli>(end - start).count();​

​

    printf("Veri Boyutu: %lu MB | ", size / (1024 * 1024));​

    printf("H2D Süresi: %.3f ms | D2H Süresi: %.3f ms | ", h2d_time, d2h_time);​

    printf("Pinned: %s\n", usePinnedMemory ? "EVET" : "HAYIR");​

​ // Bellek temizleme​

    if (usePinnedMemory) {​

        cudaFreeHost(h_data);​

    } else {​

        free(h_data);​

    }​

    cudaFree(d_data);​

}​

​

int main() {​

    size_t sizes[] = {1 * 1024 * 1024, 10 * 1024 * 1024, 100 * 1024 * 1024, 1024 * 1024 * 1024}; // 1MB, 10MB, 100MB, 1GB​

​

    printf("🔹 Normal (Paged Memory) Transfer Süreleri:\n");​

    for (size_t size : sizes) {​

        measure_transfer_time(size, false);​

    }​

​

    printf("\n🔹 Pinned Memory Transfer Süreleri:\n");​

    for (size_t size : sizes) {​

        measure_transfer_time(size, true);​

    }​

​

    return 0;​

}​
